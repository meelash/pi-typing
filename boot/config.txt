# Typing Adventure - Raspberry Pi 3 B+ boot configuration
# Read by the GPU firmware before our kernel starts.

# Run our bare-metal kernel in 64-bit mode.
arm_64bit=1
kernel=kernel8.img
kernel_address=0x80000
initial_turbo=0

# There is no operating system, so no device tree is loaded (and no .dtb
# files are on the card). This also means no drivers or overlays exist here.
device_tree=

# Force HDMI mode (not DVI) so sound works through HDMI speakers,
# and use the whole screen.
hdmi_drive=2
disable_overscan=1
disable_splash=1
boot_delay=0
